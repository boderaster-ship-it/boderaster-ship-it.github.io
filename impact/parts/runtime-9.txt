/* v5 regression hotfix: camera input, camera framing and non-drone soundscape */
AudioEngine.prototype.startAmbience=function(){
  /* Deliberately no continuous oscillator/noise bed. Silence is part of the sound design. */
  this.ambient=null;
  this.nextAmbient=performance.now()/1000+5+Math.random()*4;
};
AudioEngine.prototype.setChapter=function(id){
  this.chapter=id||1;
  this.nextAmbient=performance.now()/1000+4+Math.random()*5;
};
AudioEngine.prototype.ambientTick=function(t){
  if(!save.settings.sound||!this.ctx||t<this.nextAmbient)return;
  this.nextAmbient=t+8+Math.random()*10;
  if(this.chapter===1){
    this.tone(620+Math.random()*180,.18,'sine',.0045,820+Math.random()*180);
    if(Math.random()>.55)this.metalHit(720+Math.random()*180,.0025,.08);
  }else if(this.chapter===2){
    this.noiseBurst(.10,.0045,950);
    if(Math.random()>.55)this.metalHit(150+Math.random()*80,.004,.04);
  }else{
    this.tone(260+Math.random()*130,.30,'sine',.004,190+Math.random()*60);
    this.tone(650+Math.random()*250,.12,'sine',.0025,520,.10);
  }
};

ImpactGame.prototype.pinchMidpoint=function(){
  const a=[...this.pointers.values()];
  if(a.length<2)return{x:0,y:0};
  return{x:(a[0].x+a[1].x)/2,y:(a[0].y+a[1].y)/2};
};
ImpactGame.prototype.bindTouch=function(){
  const c=this.canvas;
  c.style.touchAction='none';
  const beginPinch=()=>{
    const m=this.pinchMidpoint();
    this.pinchStart={distance:Math.max(24,this.pinchDistance()),camDist:this.cam.distD,midX:m.x,midY:m.y,target:this.cam.desired.clone()};
  };
  c.addEventListener('pointerdown',e=>{
    if(this.screen!=='game')return;
    e.preventDefault();
    c.setPointerCapture?.(e.pointerId);
    this.pointers.set(e.pointerId,{x:e.clientX,y:e.clientY,lastX:e.clientX,lastY:e.clientY,startX:e.clientX,startY:e.clientY,moved:0});
    if(this.pointers.size===2)beginPinch();
  },{passive:false});
  c.addEventListener('pointermove',e=>{
    if(!this.pointers.has(e.pointerId)||this.screen!=='game')return;
    e.preventDefault();
    const p=this.pointers.get(e.pointerId),dx=e.clientX-p.lastX,dy=e.clientY-p.lastY;
    p.x=e.clientX;p.y=e.clientY;p.moved=Math.max(p.moved,Math.hypot(e.clientX-p.startX,e.clientY-p.startY));p.lastX=e.clientX;p.lastY=e.clientY;
    if(this.pointers.size===1){
      if(p.moved>3){
        this.cam.yawD-=dx/Math.max(320,innerWidth)*3.15;
        this.cam.pitchD=clamp(this.cam.pitchD+dy/Math.max(260,innerHeight)*2.05,-.18,1.34);
      }
    }else if(this.pointers.size===2){
      if(!this.pinchStart)beginPinch();
      const d=Math.max(24,this.pinchDistance()),target=this.pinchStart.camDist*(this.pinchStart.distance/d);
      this.cam.distD=clamp(target,this.cam.min,this.cam.max);
      const m=this.pinchMidpoint(),mx=(m.x-this.pinchStart.midX)/Math.max(320,innerWidth),my=(m.y-this.pinchStart.midY)/Math.max(260,innerHeight);
      const cp=Math.cos(this.cam.pitchD),sp=Math.sin(this.cam.pitchD);
      const offset=new THREE.Vector3(cp*Math.cos(this.cam.yawD),sp,cp*Math.sin(this.cam.yawD));
      const forward=offset.multiplyScalar(-1).normalize(),right=new THREE.Vector3().crossVectors(forward,new THREE.Vector3(0,1,0)).normalize(),up=new THREE.Vector3().crossVectors(right,forward).normalize();
      const scale=this.pinchStart.camDist*1.05;
      this.cam.desired.copy(this.pinchStart.target).addScaledVector(right,-mx*scale).addScaledVector(up,my*scale);
    }
  },{passive:false});
  const end=e=>{
    const p=this.pointers.get(e.pointerId),wasCount=this.pointers.size;
    if(p&&wasCount===1&&p.moved<7)this.pickSurface(e.clientX,e.clientY);
    this.pointers.delete(e.pointerId);
    try{c.releasePointerCapture?.(e.pointerId)}catch{}
    if(this.pointers.size===1){
      const q=[...this.pointers.values()][0];q.lastX=q.x;q.lastY=q.y;q.startX=q.x;q.startY=q.y;q.moved=0;
    }
    this.pinchStart=null;
  };
  c.addEventListener('pointerup',end,{passive:false});
  c.addEventListener('pointercancel',end,{passive:false});
  c.addEventListener('wheel',e=>{if(this.screen!=='game')return;e.preventDefault();this.cam.distD=clamp(this.cam.distD*Math.exp(e.deltaY*.001),this.cam.min,this.cam.max)},{passive:false});
};

ImpactGame.prototype.fitCamera=function(immediate=false,resizeOnly=false){
  if(!this.current)return;
  const b=new THREE.Box3();
  const add=(p,rad=0)=>{const v=v3(p);b.expandByPoint(v.clone().addScalar(rad));b.expandByPoint(v.clone().addScalar(-rad))};
  add(this.current.launcher.pos,.7);add(this.current.goal.pos,this.current.goal.r);
  for(const s of this.current.surfaces||[]){const p=v3(s.pos),half=v3(s.size).multiplyScalar(.5);b.expandByPoint(p.clone().sub(half));b.expandByPoint(p.clone().add(half));if(s.socket)add(s.socket.pos,.9)}
  for(const g of this.current.gates||[])add(g.pos,Math.max(...g.size)/2);
  for(const h of this.current.hazards||[])add(h.pos,Math.max(...h.size)/2);
  const center=b.getCenter(new THREE.Vector3()),size=b.getSize(new THREE.Vector3()),half=size.clone().multiplyScalar(.5);
  const cp=Math.cos(this.cam.pitchD),sp=Math.sin(this.cam.pitchD),offset=new THREE.Vector3(cp*Math.cos(this.cam.yawD),sp,cp*Math.sin(this.cam.yawD));
  const forward=offset.clone().multiplyScalar(-1).normalize(),right=new THREE.Vector3().crossVectors(forward,new THREE.Vector3(0,1,0)).normalize(),up=new THREE.Vector3().crossVectors(right,forward).normalize();
  let ex=0,ey=0,depth=0;
  for(const sx of [-1,1])for(const sy of [-1,1])for(const sz of [-1,1]){
    const r=new THREE.Vector3(half.x*sx,half.y*sy,half.z*sz);ex=Math.max(ex,Math.abs(r.dot(right)));ey=Math.max(ey,Math.abs(r.dot(up)));depth=Math.max(depth,Math.abs(r.dot(forward)));
  }
  const vfov=THREE.MathUtils.degToRad(this.camera.fov),hfov=2*Math.atan(Math.tan(vfov/2)*this.camera.aspect);
  let final=Math.max(ex/Math.tan(hfov/2),ey/Math.tan(vfov/2))+depth*.55+1.2;
  const portrait=innerHeight>innerWidth,dockCollapsed=$('#dock').classList.contains('collapsed');
  final*=dockCollapsed?1.03:(portrait?1.16:1.08);
  this.cam.desired.copy(center).addScaledVector(up,portrait&&!dockCollapsed?-ey*.16:-ey*.08);
  this.cam.fit=final;this.cam.min=Math.max(2.8,final*.28);this.cam.max=Math.max(55,final*5.2);
  if(!resizeOnly)this.cam.distD=clamp(final,this.cam.min,this.cam.max);
  if(immediate){this.cam.target.copy(this.cam.desired);this.cam.dist=this.cam.distD}
};
ImpactGame.prototype.resize=function(){
  const w=innerWidth,h=innerHeight;this.renderer.setSize(w,h,false);this.camera.aspect=w/h;this.camera.updateProjectionMatrix();
  if(this.current&&this.screen==='game')this.fitCamera(false,false);
};

new ImpactGame();

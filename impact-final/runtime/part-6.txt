beginCharge(){if(this.running)return;this.cannonAim.charging=true;this.cannonAim.chargeStart=performance.now();$('#fireBtn').classList.add('charging')}
releaseShot(){if(!this.cannonAim.charging)return;const held=(performance.now()-this.cannonAim.chargeStart)/1000;this.cannonAim.power=clamp(.22+held/1.25,.22,1);this.fire();this.cannonAim.charging=false;$('#fireBtn').classList.remove('charging')}
fire(){this.resetCore();this.running=true;const dir=this.getCannonDirection(),muzzle=new THREE.Vector3(1.45,.52,0).applyEuler(new THREE.Euler(0,this.cannonAim.yaw,this.cannonAim.pitch,'YXZ')).add(this.cannon.position);this.core.position.copy(muzzle);this.coreVelocity.copy(dir.multiplyScalar(8+10*this.cannonAim.power));this.audio.launch(this.cannonAim.power)}
getCannonDirection(){return new THREE.Vector3(1,0,0).applyEuler(new THREE.Euler(0,this.cannonAim.yaw,this.cannonAim.pitch,'YXZ')).normalize()}
updateCannon(){if(!this.cannon)return;this.cannon.rotation.y=this.cannonAim.yaw;this.cannon.turret.rotation.z=this.cannonAim.pitch;$('#aimReadout').textContent=`YAW ${Math.round(THREE.MathUtils.radToDeg(this.cannonAim.yaw))}° • PITCH ${Math.round(THREE.MathUtils.radToDeg(this.cannonAim.pitch))}°`;this.updateTrajectory()}
updateTrajectory(){if(!this.trajectoryLine)return;if(this.mode!=='aim'){this.trajectoryLine.visible=false;return}this.trajectoryLine.visible=true;const pts=[],p0=this.cannon.position.clone().add(new THREE.Vector3(1.45,.52,0).applyEuler(new THREE.Euler(0,this.cannonAim.yaw,this.cannonAim.pitch,'YXZ')));let p=p0.clone(),v=this.getCannonDirection().multiplyScalar(8+10*(this.cannonAim.power||.45));for(let i=0;i<40;i++){pts.push(p.clone());v.y-=.18;p=p.clone().add(v.clone().multiplyScalar(.09))}this.trajectoryLine.geometry.dispose();this.trajectoryLine.geometry=new THREE.BufferGeometry().setFromPoints(pts)}
onPointerDown(e){if(this.screen!=='game'||e.target.closest('.panel,button'))return;this.audio.resume();this.pointers.set(e.pointerId,{x:e.clientX,y:e.clientY,sx:e.clientX,sy:e.clientY});this.tapMoved=false;if(this.pointers.size===2){this.dragMode='pinch';this.pinchStart=this.currentPinchDist();this.pinchRadius=this.orbit.radius;return}this.dragMode=this.mode==='aim'?'aim':'orbit'}
onPointerMove(e){if(!this.pointers.has(e.pointerId))return;const p=this.pointers.get(e.pointerId),dx=e.clientX-p.x,dy=e.clientY-p.y;p.x=e.clientX;p.y=e.clientY;if(Math.abs(e.clientX-p.sx)+Math.abs(e.clientY-p.sy)>6)this.tapMoved=true;if(this.dragMode==='pinch'&&this.pointers.size>=2){this.orbit.radius=clamp(this.pinchRadius*(this.pinchStart/this.currentPinchDist()),9,44);return}if(this.dragMode==='orbit'){this.orbit.yaw-=dx*.0085;this.orbit.pitch=clamp(this.orbit.pitch-dy*.0065,.18,1.22)}else if(this.dragMode==='aim'){this.cannonAim.yaw-=dx*.008;this.cannonAim.pitch=clamp(this.cannonAim.pitch-dy*.005,-.18,.62);this.updateCannon()}}
onPointerUp(e){if(!this.pointers.has(e.pointerId))return;const p=this.pointers.get(e.pointerId),tap=!this.tapMoved&&Math.abs(e.clientX-p.sx)+Math.abs(e.clientY-p.sy)<8;this.pointers.delete(e.pointerId);if(tap&&this.mode==='edit'&&!this.running)this.pickMachine(e.clientX,e.clientY);if(this.pointers.size===0)this.dragMode=null}
currentPinchDist(){const a=[...this.pointers.values()];return a.length<2?0:Math.hypot(a[0].x-a[1].x,a[0].y-a[1].y)}
pickMachine(x,y){const r=this.canvas.getBoundingClientRect();this.pointer.set(((x-r.left)/r.width)*2-1,-((y-r.top)/r.height)*2+1);this.raycaster.setFromCamera(this.pointer,this.camera);const hits=this.raycaster.intersectObjects(this.machines.map(m=>m.group),true);if(!hits.length)return;let o=hits[0].object;while(o&&!o.userData.machine)o=o.parent;if(o?.userData.machine)this.selectMachine(o.userData.machine)}
